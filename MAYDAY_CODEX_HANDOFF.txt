          import concurrent.futures


          class ExecutionEngine:
              def __init__(self):
                  self._tools = {}
              def register_tools(self, tools: dict):
                  self._tools = tools
              def execute(self, tool_name: str, parameters: dict) -> dict:
                  base = tool_name.split('_')[0]
                  tool = self._tools.get(base)
                  if not tool:
                      return {'status': 'error',
                              'error': f'Unknown tool: {tool_name}'}
                  params = {**parameters, '_tool_name': tool_name}
                  with concurrent.futures.ThreadPoolExecutor(max_workers=1) as ex:
                      fut = ex.submit(tool.execute, params)
                      try:
                          return fut.result(timeout=30)
                      except concurrent.futures.TimeoutError:
                          raise ToolTimeoutError(
                              f'{tool_name} exceeded 30s timeout')

  T00.7b Create scripts/bootstrap_check.py (self-heal brain):


          """
          Portable health check. Called by build.bat with one of:
            --install-runtime   download embeddable Python to runtime/python/
            --check-deps        exit 1 if any required package missing
            --check-models      exit 1 if models/ empty or missing
          """
          import sys, os, subprocess, urllib.request, zipfile, importlib.util
          from pathlib import Path


          ROOT = Path(__file__).resolve().parent.parent
          RUNTIME_DIR = ROOT / 'runtime' / 'python'
          MODELS_DIR = ROOT / 'models'
          PY_EMBED_URL = ('https://www.python.org/ftp/python/3.10.11/'
                          'python-3.10.11-embed-amd64.zip')


          def install_runtime():
              RUNTIME_DIR.mkdir(parents=True, exist_ok=True)
              zip_path = ROOT / 'python_embed.zip'
              urllib.request.urlretrieve(PY_EMBED_URL, zip_path)
              with zipfile.ZipFile(zip_path) as z:
                  z.extractall(RUNTIME_DIR)
              zip_path.unlink()
              for pth in RUNTIME_DIR.glob('python*._pth'):
                  txt = pth.read_text()
                  pth.write_text(txt.replace('#import site', 'import site'))
              return 0


          def check_deps():
              required = ['PyQt6', 'torch', 'transformers',
                          'peft', 'httpx', 'playwright', 'cryptography',
                          'huggingface_hub', 'psutil']
              missing = [m for m in required
                         if importlib.util.find_spec(m) is None]
              return 1 if missing else 0


          def check_models():
              if not MODELS_DIR.exists():
                  return 1
              files = [f for f in MODELS_DIR.rglob('*')
                       if f.is_file() and f.stat().st_size > 1024 * 1024]
              return 0 if files else 1


          if __name__ == '__main__':
              arg = sys.argv[1] if len(sys.argv) > 1 else ''
              if arg == '--install-runtime': sys.exit(install_runtime())
              if arg == '--check-deps':     sys.exit(check_deps())
              if arg == '--check-models':   sys.exit(check_models())
              print('Usage: bootstrap_check.py [--install-runtime|--check-deps|--check-models]')
              sys.exit(2)

           summarize_and_promote(threshold=30):
             when short_term rows for session > threshold:
               * take oldest (count - 10) rows
               * call ContextCompressor.summarize(rows) → summary
               * insert into mid_term with source_ids
               * delete those rows from short_term
           full_context(token_budget=2000) -> str:
             interleave: long_term (relevant) → mid_term (last N)
             → short_term (recent), trimming to budget


  T07b.3 runtime/task_memory.py:

           create_task(title, goal_id=None, deps=[]) -> id
           update_status(task_id, status)   # pending|active|done|failed

           dependencies_met(task_id) -> bool

           create_goal(title, parent_id=None) -> id
           list_active() -> list

           summarize(rows: list[dict]) -> str

               keeping decisions, file paths, errors."
             - Calls local model via ModelRouter (NEVER external API
               unless user_approved per R5)
             - Returns plain text summary
           compress_to_long(session_id):
             - Aggregate mid_term entries → semantic facts
             - Insert into long_term with key='session:{id}:facts'
